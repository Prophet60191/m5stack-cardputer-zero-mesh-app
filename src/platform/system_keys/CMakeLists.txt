if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 cmake_minimum_required(VERSION 3.31)
 project(cp0_system_keys LANGUAGES CXX)
 set(CMAKE_CXX_STANDARD 17)
 set(CMAKE_CXX_STANDARD_REQUIRED ON)
 include(CTest)
endif()
# SPDX-License-Identifier: MIT
find_package(Threads REQUIRED)
set(cp0_key_sources system_keys.cpp upstream/cp0_audio_mixer_control.cpp upstream/cp0_pulse_client.cpp)
add_library(cp0_system_keys STATIC ${cp0_key_sources})
set_target_properties(cp0_system_keys PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(cp0_system_keys PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(cp0_system_keys PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
option(CP0_KEYS_BUILD_SHARED "Build a compatibility library for Python apps" OFF)
if(CP0_KEYS_BUILD_SHARED)
 add_library(cp0_system_keys_shared SHARED ${cp0_key_sources})
 set_target_properties(cp0_system_keys_shared PROPERTIES OUTPUT_NAME cp0-system-keys)
 target_link_libraries(cp0_system_keys_shared PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
endif()
if(BUILD_TESTING)
 add_executable(cp0_system_keys_tests tests.cpp)
 target_link_libraries(cp0_system_keys_tests PRIVATE cp0_system_keys)
 add_test(NAME cp0_system_keys_tests COMMAND cp0_system_keys_tests)
endif()
